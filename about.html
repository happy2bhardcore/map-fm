function timeAgo(uts) {
  if (!uts) return "unknown time ago";
  const seconds = Math.floor((Date.now() / 1000) - Number(uts));
  if (seconds < 60) return `${Math.max(1, seconds)} seconds ago`;
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} minute${minutes === 1 ? '' : 's'} ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours} hour${hours === 1 ? '' : 's'} ago`;
  const days = Math.floor(hours / 24);
  return `${days} day${days === 1 ? '' : 's'} ago`;
}

recentData = {
  searchControl.trigger;
  if (!username) return null;
  const apiKey = "b914193d6c62aabcd83adf2b6c457a5f";
  const url = `https://ws.audioscrobbler.com/2.0/?method=user.getrecenttracks&user=${encodeURIComponent(username)}&api_key=${apiKey}&format=json&limit=1&_t=${Date.now()}`;
  const response = await fetch(url);

track = {
  if (!recentData || !recentData.recenttracks || !recentData.recenttracks.track) return null;
  const t = recentData.recenttracks.track;
  return Array.isArray(t) ? t[0] : t;
}

trackInfo = {
  if (!username || !track) return null;
  const apiKey = "b914193d6c62aabcd83adf2b6c457a5f";
  const songName = track.name || "";
  const artistName = track.artist ? (track.artist["#text"] || track.artist.name) : "";
  if (!songName || !artistName) return null;
  try {
    const url = `https://ws.audioscrobbler.com/2.0/?method=track.getInfo&artist=${encodeURIComponent(artistName)}&track=${encodeURIComponent(songName)}&username=${encodeURIComponent(username)}&api_key=${apiKey}&format=json&_t=${Date.now()}`;
    const response = await fetch(url);
    return await response.json();
  } catch (e) {
    return null;
  }
}
