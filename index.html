This page lets you look up a Last.fm user and see their top 9 most listened to albums.

viewof searchControl = {
  const form = html`
    <form style="max-width: 440px; margin-bottom: 1.25rem;">
      <label style="display: block; font-weight: 600; font-size: 0.95rem; margin-bottom: 6px;">Last.fm Username:</label>
      <div style="display: flex; gap: 8px; align-items: center;">
        <input type="text" placeholder="e.g. rj" class="form-control" style="flex: 1; height: 38px; border-radius: 6px;" />
        <button type="submit" class="btn btn-primary" style="height: 38px; padding: 0 16px; border-radius: 6px; font-weight: 500;">Search</button>
        <button type="button" class="btn btn-outline-secondary" title="Refresh" style="height: 38px; width: 38px; min-width: 38px; padding: 0; display: inline-flex; align-items: center; justify-content: center; border-radius: 6px;">
          <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" fill="currentColor" viewBox="0 0 16 16">
            <path fill-rule="evenodd" d="M8 3a5 5 0 1 0 4.546 2.914.5.5 0 0 1 .908-.417A6 6 0 1 1 8 2z"/>
            <path d="M8 4.466V.534a.25.25 0 0 1 .41-.192l2.36 1.966c.12.1.12.284 0 .384L8.41 4.658A.25.25 0 0 1 8 4.466"/>
          </svg>
        </button>
      </div>
    </form>
  `;

  const input = form.querySelector('input');
  const refreshBtn = form.querySelector('button[type="button"]');

  form.value = { username: "", trigger: 0 };

  form.onsubmit = (e) => {
    e.preventDefault();
    form.value = { username: input.value.trim(), trigger: form.value.trigger + 1 };
    form.dispatchEvent(new CustomEvent("input"));
  };

  refreshBtn.onclick = () => {
    form.value = { username: input.value.trim(), trigger: form.value.trigger + 1 };
    form.dispatchEvent(new CustomEvent("input"));
  };

  return form;
}

username = searchControl.username;
topAlbumsData = {
  searchControl.trigger;
  if (!username) return null;
  const apiKey = "b914193d6c62aabcd83adf2b6c457a5f";
  const url = `https://ws.audioscrobbler.com/2.0/?method=user.gettopalbums&user=${encodeURIComponent(username)}&api_key=${apiKey}&format=json&limit=9&_t=${Date.now()}`;
  const response = await fetch(url);
  return await response.json();
}

albums = {
  if (!topAlbumsData || !topAlbumsData.topalbums || !topAlbumsData.topalbums.album) return null;
  const list = topAlbumsData.topalbums.album;
  return Array.isArray(list) ? list : [list];
}

html`${
  !username ? html`<em>Enter a Last.fm username above to see their top albums.</em>` :
  topAlbumsData?.error ? html`<div class="alert alert-danger" style="margin-top: 10px;">❌ ${topAlbumsData.message}</div>` :
  !albums || albums.length === 0 ? html`<div class="alert alert-warning" style="margin-top: 10px;">No top albums found for this user.</div>` :
  (() => {
    return html`
      <div style="margin-top: 20px;">
        <h4 style="margin-bottom: 16px; font-weight: 600;">Top 9 Albums for ${username}</h4>
        <div style="display: grid; grid-template-columns: repeat(auto-fill, minmax(180px, 1fr)); gap: 16px;">
          ${albums.slice(0, 9).map((album, idx) => {
            const images = album.image;
            let imageUrl = "";
            if (Array.isArray(images)) {
              const preferred = images.find(img => img.size === "extralarge") ||
                                images.find(img => img.size === "large") ||
                                images[images.length - 1];
              imageUrl = preferred ? preferred["#text"] : "";
            } else if (images && images["#text"]) {
              imageUrl = images["#text"];
            }

            const albumName = album.name || "Unknown Album";
            const artistName = album.artist ? album.artist.name : "Unknown Artist";
            const playcount = album.playcount ? Number(album.playcount).toLocaleString() : "0";

            return html`
              <div class="card h-100" style="border: 1px solid #ddd; border-radius: 8px; overflow: hidden; display: flex; flex-direction: column;">
                ${imageUrl ? html`
                  <img src="${imageUrl}" alt="${albumName}" style="width: 100%; aspect-ratio: 1; object-fit: cover; display: block;" />
                ` : html`
                  <div style="width: 100%; aspect-ratio: 1; background: #eee; display: flex; align-items: center; justify-content: center; color: #888; font-size: 0.9rem;">
                    No Image
                  </div>
                `}
                <div style="padding: 10px; display: flex; flex-direction: column; flex-grow: 1;">
                  <div style="font-weight: 600; font-size: 0.95rem; line-height: 1.2; margin-bottom: 4px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;" title="${albumName}">${albumName}</div>
                  <div style="font-size: 0.85rem; color: #555; margin-bottom: 6px; overflow: hidden; text-overflow: ellipsis; white-space: nowrap;" title="${artistName}">${artistName}</div>
                  <div style="font-size: 0.75rem; color: #888; margin-top: auto;">${playcount} plays</div>
                </div>
              </div>
            `;
          })}
        </div>
      </div>
    `;
  })()
}`
