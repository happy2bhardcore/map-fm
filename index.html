        <div style="font-size: 0.8rem; color: #ffffff; margin-bottom: 4px; text-shadow: 0 2px 8px rgba(0,0,0,0.95);">scrobbled ${playcount} times</div>
        <div class="fav-track-line" style="font-size: 0.78rem; color: #ffffff; line-height: 1.35; text-shadow: 0 2px 8px rgba(0,0,0,0.95);">
          favorite track: <span style="font-style: italic; color: #ddd;">finding...</span>
        </div>
      </div>
    `;

    nodeEl.appendChild(info);

    getFavoriteTrack(artistName, albumName).then((fav) => {
      if (!nodeEl.classList.contains("album-focused")) return;
      const line = info.querySelector(".fav-track-line");
      if (!line) return;
      if (fav && fav.name) {
        line.innerHTML = `favorite track: ${fav.name} (scrobbled ${parseInt(fav.count || 0, 10).toLocaleString()} times)`;
      } else {
        line.innerHTML = `favorite track: None found (scrobbled 0 times)`;
      }
    });
  }

  function deactivateFocus(node, nodeEl) {
    if (focusedItem && focusedItem.el === nodeEl) {
      focusedItem = null;
    }
    if (backdropOverlay) backdropOverlay.style.opacity = "0";
    nodeEl.classList.remove("album-focused");
    const info = nodeEl.querySelector(".album-focus-info");
    if (info) info.remove();
  }

  function addNextNode() {
    if (nextAlbumIndex >= verifiedAlbums.length) return;
    const item = verifiedAlbums[nextAlbumIndex++];
    addNode(item);
  }

  function addNode(item) {
    const { album, url: imageUrl } = item;
    const albumName = album.name || "Unknown Album";
    const artistName = album.artist ? (album.artist.name || album.artist["#text"] || album.artist) : "Unknown Artist";

    const angle = (nodes.length * 0.6) + (Math.random() - 0.5) * 0.4;
    const dist = 25 + Math.random() * 20;

    const node = {
      id: nodes.length,
      x: width / 2 + Math.cos(angle) * dist,
      y: height / 2 + Math.sin(angle) * dist,
      vx: (Math.random() - 0.5) * 2,
      vy: (Math.random() - 0.5) * 2
    };

    const nodeEl = html`
      <div class="album-node" style="position: absolute; top: 0; left: 0; width: ${nodeSize}px; height: ${nodeSize}px; cursor: grab; will-change: transform;" title="${albumName} - ${artistName}">
        <div class="album-card-inner">
        </div>
      </div>
    `;

    const inner = nodeEl.querySelector(".album-card-inner");
    const isGif = (u) => /\.gif($|\?)/i.test(u);

    if (isGif(imageUrl)) {
      const canvas = html`<canvas style="width: 100%; height: 100%; object-fit: cover; display: block;"></canvas>`;
      const img = new Image();
      img.onload = () => {
        canvas.width = img.naturalWidth || nodeSize;
        canvas.height = img.naturalHeight || nodeSize;
        const ctx = canvas.getContext("2d");
        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
      };
      img.src = imageUrl;
      inner.appendChild(canvas);
    } else {
      const img = html`<img src="${imageUrl}" alt="${albumName}" style="width: 100%; height: 100%; object-fit: cover; display: block;" />`;
      inner.appendChild(img);
    }

    let hoverTimeout = null;
    let isHovering = false;
    let isDragging = false;

    nodeEl.addEventListener("mouseenter", () => {
      isHovering = true;
      hoverTimeout = setTimeout(() => {
        if (isHovering && !isDragging) {
          activateFocus(node, nodeEl, album);
        }
      }, 400);
    });

    nodeEl.addEventListener("mouseleave", () => {
      isHovering = false;
      if (hoverTimeout) {
        clearTimeout(hoverTimeout);
        hoverTimeout = null;
      }
      if (nodeEl.classList.contains("album-focused")) {
        deactivateFocus(node, nodeEl);
      }
    });

    d3.select(nodeEl).call(
      d3.drag()
        .on("start", (event) => {
          isDragging = true;
          if (hoverTimeout) {
            clearTimeout(hoverTimeout);
            hoverTimeout = null;
          }
          if (nodeEl.classList.contains("album-focused")) {
            deactivateFocus(node, nodeEl);
          }
          if (!event.active) simulation.alphaTarget(0.3).restart();
          node.fx = node.x;
          node.fy = node.y;
          nodeEl.style.cursor = "grabbing";
          nodeEl.style.zIndex = "50";
        })
        .on("drag", (event) => {
          node.fx = event.x;
          node.fy = event.y;
        })
        .on("end", (event) => {
          isDragging = false;
          if (!event.active) simulation.alphaTarget(0);
          node.fx = null;
          node.fy = null;
          nodeEl.style.cursor = "grab";
          nodeEl.style.zIndex = "";
        })
    );

    node.el = nodeEl;
    nodes.push(node);
    container.appendChild(nodeEl);
    simulation.nodes(nodes);
    simulation.alpha(0.65).restart();
  }

  function removeNode() {
    if (nodes.length === 0) return;
    const removed = nodes.pop();
    if (removed && removed.el) {
      removed.el.remove();
    }
    nextAlbumIndex = nodes.length;
  }

  let spawnTimer = null;
  container.syncCount = async (target, accelerateEntrance = false) => {
    currentTarget = target;
    if (!simulationReady) return;

    if (spawnTimer) {
      clearTimeout(spawnTimer);
      spawnTimer = null;
    }

    if (target > verifiedAlbums.length && candidateIdx < albums.length) {
      await loadMoreVerified(target);
    }

    const maxAvail = Math.min(target, verifiedAlbums.length);

    const targetSize = Math.max(26, Math.min(76, Math.round(440 / Math.sqrt(maxAvail + 4))));
    if (Math.abs(targetSize - nodeSize) >= 3) {
      nodeSize = targetSize;
      radius = nodeSize * 0.58;
      simulation.force("collide", d3.forceCollide().radius(radius).strength(1).iterations(6));
      simulation.force("charge", d3.forceManyBody().strength(-15).distanceMax(nodeSize * 2));
      for (const d of nodes) {
        if (d.el) {
          d.el.style.width = `${nodeSize}px`;
          d.el.style.height = `${nodeSize}px`;
        }
      }
    }

    if (nodes.length > maxAvail) {
      while (nodes.length > maxAvail) {
        removeNode();
      }
      simulation.nodes(nodes);
      simulation.alpha(0.4).restart();
    } else if (nodes.length < maxAvail) {
      function spawnStep() {
        if (cancelled || nodes.length >= maxAvail) {
          spawnTimer = null;
          return;
        }
        addNextNode();
        if (nodes.length >= maxAvail) {
          spawnTimer = null;
          return;
        }
        let delay;
        if (accelerateEntrance) {
          const progress = nodes.length / maxAvail;
          delay = Math.round(15 + 465 * Math.pow(1 - progress, 2.4));
        } else {
          const diff = maxAvail - nodes.length;
          delay = Math.max(10, Math.min(60, Math.round(250 / diff)));
        }
        spawnTimer = setTimeout(spawnStep, delay);
      }
      spawnStep();
    }
  };

  async function init() {
    const targetInitial = Math.min(50, albums.length);
    await loadMoreVerified(targetInitial);
    if (cancelled) return;

    const loader = container.querySelector('.chart-loading-indicator');
    if (loader) loader.remove();

    const isFirstTime = !window.__chartFmEverLoaded;
    window.__chartFmEverLoaded = true;

    simulationReady = true;
    container.syncCount(currentTarget, isFirstTime);

    // Preload remaining albums in the background so count slider changes are instant
    loadMoreVerified(Math.min(100, albums.length));
  }

  init();

  invalidation.then(() => {
    cancelled = true;
    document.body.classList.remove("has-chart");
    window.removeEventListener("resize", updateDimensions);
    if (spawnTimer) clearTimeout(spawnTimer);
    simulation.stop();
  });

  return container;
}
updateSimulation = {
  albumLimit;
  if (chartView && typeof chartView.syncCount === "function") {
    chartView.syncCount(albumLimit);
  }
